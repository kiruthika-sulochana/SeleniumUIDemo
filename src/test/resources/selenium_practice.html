# Detect broken images
images = driver.find_elements(By.TAG_NAME, "img")
for img in images:
  is_broken = driver.execute_script(
    "return arguments[0].naturalWidth === 0", img)
  if is_broken: print(f"Broken: {img.get_attribute('src')}")

Quick Reference Cheatsheet
# === LOCATOR STRATEGIES ===
By.ID | By.NAME | By.CLASS_NAME | By.TAG_NAME
By.CSS_SELECTOR | By.XPATH | By.LINK_TEXT | By.PARTIAL_LINK_TEXT

# === ELEMENT ACTIONS ===
el.click() | el.send_keys("text") | el.clear() | el.submit()
el.text | el.get_attribute("value") | el.is_displayed() | el.is_enabled()

# === WAITS ===
WebDriverWait(driver, 10).until(EC.presence_of_element_located((By.ID,"x")))
EC.element_to_be_clickable | EC.invisibility_of_element | EC.text_to_be_present

# === SELECT DROPDOWN ===
from selenium.webdriver.support.ui import Select
Select(el).select_by_visible_text("Option") | .select_by_value("v") | .select_by_index(0)

# === ACTION CHAINS ===
ActionChains(driver).move_to_element(el).click().perform()
ActionChains(driver).drag_and_drop(src, tgt).perform()
ActionChains(driver).double_click(el).perform()

# === WINDOW / FRAME ===
driver.switch_to.frame("id") | driver.switch_to.default_content()
driver.switch_to.window(handle) | driver.window_handles
driver.switch_to.alert.accept() | .dismiss() | .send_keys("text")

# === JS EXECUTOR ===
driver.execute_script("arguments[0].click()", el)
driver.execute_script("window.scrollTo(0, document.body.scrollHeight)")
driver.execute_script("return arguments[0].shadowRoot", host)
